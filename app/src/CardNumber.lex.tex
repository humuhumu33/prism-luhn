  name: isDigits,
  parameters: [value : Bytes, offset : Nat, remaining : Nat],
  result: Bool,
  body: (match remaining

  name: checksum,
  parameters: [value : Bytes, remaining : Nat, parity : Nat, accumulated : Nat],
  result: Nat,
  body: (match remaining
    | Nat.zero => accumulated
    | Nat.succ rest => (match (Foundation.Bytes::byteAt value rest)

      | Option.some byte => (checksum value rest (prim subtract 1 parity : Nat) (accumulated + (weighted (digitOf byte) parity))))),
  axioms: [Quot.sound, propext],
  recursive_argument: remaining
}

  name: isLuhn,
  parameters: [value : Bytes],
  result: Bool,
  body: ((prim remainder (checksum value (Foundation.Bytes::byteLength value) 0 0) 10 0 : Nat) =? 0),
  axioms: [Quot.sound, propext]
}

  name: answer,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if (isLuhn value) then (prim utf8_encode "valid" : Bytes) else (prim utf8_encode "invalid" : Bytes)),
  axioms: [Quot.sound, propext]
}

definition{
  name: isCardNumber,
  parameters: [value : Bytes],
  result: Bool,
  body: ((7 <? (Foundation.Bytes::byteLength value)) && (((Foundation.Bytes::byteLength value) <? 20) && (isDigits value 0 (Foundation.Bytes::byteLength value)))),
  axioms: [Quot.sound, propext]
}

  name: dispatch,

  result: Bytes,
  body: (if (isCardNumber value) then (answer value) else (prim utf8_encode "invalid request" : Bytes)),
  axioms: [Quot.sound, propext]
